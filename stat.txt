summary_stats <- function(data) {

  cat("Mean:", mean(data, na.rm = TRUE), "\n")
  cat("Median:", median(data, na.rm = TRUE), "\n")
  cat("Standard Deviation:", sd(data, na.rm = TRUE), "\n")
  cat("Variance:", var(data, na.rm = TRUE), "\n")
  cat("Minimum:", min(data, na.rm = TRUE), "\n")
  cat("Maximum:", max(data, na.rm = TRUE), "\n")
  cat("Range:", diff(range(data, na.rm = TRUE)), "\n")
  
  cat("Quartiles:\n")
  print(quantile(data, na.rm = TRUE))
  
  cat("IQR:", IQR(data, na.rm = TRUE), "\n")
  cat("Coefficient of Variation:", sd(data, na.rm = TRUE) / mean(data, na.rm = TRUE), "\n\n")
}

print("Organizations Dataset Summary")
apply(organizations_num, 2, summary_stats)

print("People Dataset Summary")
apply(people_num, 2, summary_stats)

print("Products Dataset Summary")
apply(products_num, 2, summary_stats)

cat("\n---- Z-TEST ----\n")

sample_data <- c(52, 55, 49, 50, 53, 48, 54)

sample_mean <- mean(sample_data)
sample_sd <- sd(sample_data)
n <- length(sample_data)

se <- sample_sd / sqrt(n)

z_value <- (sample_mean - 50) / se
cat("Z-value =", z_value, "\n")

p_value_z <- 2 * (1 - pnorm(abs(z_value)))
cat("p-value =", p_value_z, "\n")


cat("\n---- T-TEST ----\n")

t_test_result <- t.test(sample_data, mu = 50, conf.level = 0.95)

print(t_test_result)


cat("\n---- MANUAL CONFIDENCE INTERVAL ----\n")

t_critical <- qt(0.975, df = n - 1)

lower_ci <- sample_mean - t_critical * se
upper_ci <- sample_mean + t_critical * se

cat("95% Confidence Interval:\n")
cat("Lower =", lower_ci, "\n")
cat("Upper =", upper_ci, "\n")

cat("\n===== END =====\n")
